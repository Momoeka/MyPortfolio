  \@ifundefined{@addresstwo}{
    \def\@addresstwo{#1}
  }{
  \@ifundefined{@addressthree}{
    \def\@addressthree{#1}
  }{
    \def\@addressone{#1}

\def\printaddress#1{
  \begingroup
    \def\\{\addressSep\ }
    \centerline{#1}
  \endgroup
  \par
  \addressskip
}

\def\printname{
  \begingroup
    \hfil{\MakeUppercase{\namesize\bf \@name}}\hfil
    \nameskip\break
  \endgroup
}

% Prints the name followed by any address lines that were set
\def\printheader{
  \printname
  \@ifundefined{@addressone}{}{\printaddress{\@addressone}}
  \@ifundefined{@addresstwo}{}{\printaddress{\@addresstwo}}
  \@ifundefined{@addressthree}{}{\printaddress{\@addressthree}}
}
